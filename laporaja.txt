<?php

namespace App\Http\Controllers;

use App\Models\Laporan;
use App\Models\Komplain;
use App\Models\StatusLaporan;
use App\Models\User;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\Hash;
use Illuminate\Support\Facades\Storage;

class AdminController extends Controller
{
    // ============================================
    // VERIFIKASI LAPORAN (SUPERVISOR)
    // ============================================
    public function verifikasiIndex()
    {
        $laporans = Laporan::where('moderation_status', 'pending')
            ->where('status', 'pending')
            ->latest()->get();

        return view('admin.verifikasi', compact('laporans'));
    }

    public function verifikasi(Request $request, Laporan $laporan)
    {
        // Blokir kalau staff sudah mulai proses
        if ($laporan->status !== 'pending') {
            return back()->with('error',
                'Laporan sudah diproses staff. Verifikasi tidak bisa dilakukan lagi.'
            );
        }

        $data = $request->validate([
            'moderation_status' => 'required|in:approved,rejected',
            'alasan_moderasi'   => 'nullable|string|max:500',
        ]);

        if ($data['moderation_status'] === 'rejected' && empty($data['alasan_moderasi'])) {
            return back()->with('error', 'Alasan penolakan wajib diisi.');
        }

        $laporan->update([
            'moderation_status' => $data['moderation_status'],
            'alasan_moderasi'   => $data['alasan_moderasi'] ?? null,
            'verified_by'       => auth()->id(),
            'verified_at'       => now(),
        ]);

        if ($data['moderation_status'] === 'rejected') {
            $laporan->update(['status' => 'ditolak']);
        }

        StatusLaporan::create([
            'laporan_id' => $laporan->id,
            'user_id'    => auth()->id(),
            'status'     => $laporan->status,
            'keterangan' => $data['moderation_status'] === 'approved'
                ? 'Laporan diverifikasi oleh admin.'
                : 'Laporan ditolak admin: ' . $data['alasan_moderasi'],
        ]);

        return back()->with('success', 'Verifikasi laporan berhasil.');
    }

    // ============================================
    // UPDATE STATUS (Admin & Staff)
    // ============================================
    public function updateStatus(Request $request, Laporan $laporan)
    {
        $rules = [
            'status'  => 'required|in:pending,diproses,selesai,ditolak',
            'catatan' => 'nullable|string|max:1000',
        ];

        if ($request->status === 'selesai') {
            $rules['foto_perbaikan'] = 'required|image|mimes:jpg,jpeg,png|max:2048';
        } else {
            $rules['foto_perbaikan'] = 'nullable|image|mimes:jpg,jpeg,png|max:2048';
        }

        $data = $request->validate($rules, [
            'foto_perbaikan.required' => 'Foto bukti perbaikan wajib diunggah saat status Selesai.',
        ]);

        $update = [
            'status'        => $data['status'],
            'staff_id'      => auth()->id(),
            'catatan_staff' => $data['catatan'] ?? null,
        ];

        if ($request->hasFile('foto_perbaikan')) {
            if ($laporan->foto_perbaikan && Storage::disk('public')->exists($laporan->foto_perbaikan)) {
                Storage::disk('public')->delete($laporan->foto_perbaikan);
            }
            $update['foto_perbaikan'] = $request->file('foto_perbaikan')->store('perbaikan', 'public');
        }

        $laporan->update($update);

        StatusLaporan::create([
            'laporan_id' => $laporan->id,
            'user_id'    => auth()->id(),
            'status'     => $data['status'],
            'keterangan' => $data['catatan'] ?? 'Status diperbarui.',
        ]);

        return back()->with('success', 'Status laporan berhasil diperbarui.');
    }

    // ============================================
    // KOMPLAIN — ADMIN RESPON
    // ============================================
    public function komplainIndex()
    {
        $komplains = Komplain::latest()->get();
        return view('admin.komplain', compact('komplains'));
    }

    public function komplainRespon(Request $request, Komplain $komplain)
    {
        $data = $request->validate([
            'respon_admin' => 'required|string|max:1000',
            'aksi'         => 'required|in:respon_saja,tolak_laporan,hapus_laporan',
        ], [
            'aksi.required' => 'Pilih aksi terlebih dahulu.',
        ]);

        // Update komplain dulu
        $komplain->update([
            'respon_admin' => $data['respon_admin'],
            'admin_id'     => auth()->id(),
            'respon_at'    => now(),
            'status'       => 'direspon',
        ]);

        // Handle aksi
        if ($data['aksi'] === 'tolak_laporan') {
            $komplain->laporan->update([
                'status'            => 'ditolak',
                'moderation_status' => 'rejected',
                'alasan_moderasi'   => 'Ditolak atas permintaan staff: ' . $data['respon_admin'],
                'verified_by'       => auth()->id(),
                'verified_at'       => now(),
            ]);

            StatusLaporan::create([
                'laporan_id' => $komplain->laporan_id,
                'user_id'    => auth()->id(),
                'status'     => 'ditolak',
                'keterangan' => 'Laporan ditolak atas permintaan staff.',
            ]);

            return back()->with('success', 'Laporan ditolak. Tetap tersimpan di histori.');

        } elseif ($data['aksi'] === 'hapus_laporan') {
            $laporan = $komplain->laporan;

            if ($laporan) {
                if ($laporan->foto_bukti && Storage::disk('public')->exists($laporan->foto_bukti)) {
                    Storage::disk('public')->delete($laporan->foto_bukti);
                }
                if ($laporan->foto_perbaikan && Storage::disk('public')->exists($laporan->foto_perbaikan)) {
                    Storage::disk('public')->delete($laporan->foto_perbaikan);
                }

                $laporan->komplains()->delete();
                $laporan->statusLogs()->delete();
                $laporan->delete();
            }

            return back()->with('success', 'Laporan berhasil dihapus permanen.');
        }

        return back()->with('success', 'Respon komplain berhasil dikirim.');
    }

    // ============================================
    // KELOLA USER
    // ============================================
    public function userIndex()
    {
        $users = User::where('role', '!=', 'super_admin')
            ->latest()->get();

        return view('admin.users', compact('users'));
    }

    /**
     * Tambah user baru (Admin bisa tambah user, staff, atau admin)
     * - User & Staff → langsung approved
     * - Admin → pending, tunggu approval Super Admin
     */
    public function userStore(Request $request)
    {
        $data = $request->validate([
            'name'       => 'required|string|max:100',
            'email'      => 'required|email|unique:users,email',
            'password'   => 'required|min:6',
            'role'       => 'required|in:user,staff,admin',
            'nis_nip'    => 'nullable|string',
            'kelas'      => 'nullable|string',
            'no_telepon' => 'nullable|string',
        ]);

        $data['password'] = Hash::make($data['password']);

        // Kalau admin → perlu approval Super Admin dulu
        if ($data['role'] === 'admin') {
            $data['status_akun'] = 'pending';
            $data['is_approved'] = false;

            User::create($data);

            return back()->with('success',
                "Admin {$data['name']} berhasil diajukan. Menunggu approval Super Admin."
            );
        }

        // User & Staff → langsung aktif
        $data['status_akun'] = 'approved';
        $data['is_approved'] = true;

        User::create($data);

        return back()->with('success', "User {$data['name']} berhasil ditambahkan.");
    }

    public function userDestroy(User $user)
    {
        if ($user->role === 'super_admin') {
            return back()->with('error', 'Super Admin tidak dapat dihapus.');
        }
        $user->delete();
        return back()->with('success', 'User berhasil dihapus.');
    }
}

<?php

namespace App\Http\Controllers;

use App\Models\User;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\Auth;
use Illuminate\Support\Facades\Hash;

class AuthController extends Controller
{
    public function showLogin()
    {
        return view('auth.login');
    }

    public function showRegister()
    {
        return view('auth.register');
    }

    public function register(Request $request)
    {
        $data = $request->validate([
            'name'             => 'required|string|max:100',
            'email'            => 'required|email|unique:users,email',
            'password'         => 'required|min:6|confirmed',
            'nis_nip'          => 'nullable|string|max:30',
            'kelas'            => 'nullable|string|max:30',
            'no_telepon'       => 'nullable|string|max:20',
            'alasan_pengajuan' => 'nullable|string|max:500',
        ]);

        $data['password']    = Hash::make($data['password']);
        $data['role']        = 'user';
        $data['is_approved'] = false;
        $data['status_akun'] = 'pending';

        User::create($data);

        return redirect()->route('login')
            ->with('success', 'Pengajuan akun berhasil dikirim! Tunggu konfirmasi dari Super Admin.');
    }

    public function login(Request $request)
    {
        $credentials = $request->validate([
            'email'    => 'required|email',
            'password' => 'required',
        ]);

        // Cek dulu apakah user ada & password benar
        $user = User::where('email', $credentials['email'])->first();

        if (!$user || !Hash::check($credentials['password'], $user->password)) {
            return back()->withErrors(['email' => 'Email atau password salah.'])->onlyInput('email');
        }

        // Cek status akun
        if ($user->status_akun === 'pending') {
            return back()->withErrors([
                'email' => 'Akun Anda masih menunggu konfirmasi Super Admin.',
            ])->onlyInput('email');
        }

        if ($user->status_akun === 'rejected') {
            $msg = 'Pengajuan akun Anda ditolak.';
            if ($user->catatan_penolakan) {
                $msg .= ' Alasan: ' . $user->catatan_penolakan;
            }
            return back()->withErrors(['email' => $msg])->onlyInput('email');
        }

        if (Auth::attempt($credentials)) {
            $request->session()->regenerate();
            return redirect()->intended(route('dashboard'));
        }

        return back()->withErrors(['email' => 'Email atau password salah.'])->onlyInput('email');
    }

    public function logout(Request $request)
    {
        Auth::logout();
        $request->session()->invalidate();
        $request->session()->regenerateToken();
        return redirect()->route('login');
    }
}

    public function index()
    {
        $user = auth()->user();

        if ($user->isUser()) {
            $laporans = Laporan::where('user_id', $user->id)
                ->latest()->get();
        } elseif ($user->isStaff()) {
            // ✅ STAFF LIHAT SEMUA LAPORAN PENDING & DIPROSES

            $laporans = Laporan::whereIn('status', ['pending', 'diproses'])

            // Admin & Super Admin
            $laporans = Laporan::latest()->get();
        }

        $stats = [
            'total'             => Laporan::count(),
            'pending'           => Laporan::where('status', 'pending')->count(),
            'diproses'          => Laporan::where('status', 'diproses')->count(),
            'selesai'           => Laporan::where('status', 'selesai')->count(),
            'perlu_verifikasi'  => Laporan::where('status', 'pending')
                                            ->where('moderation_status', 'pending')
                                            ->count(),
            'komplain_pending'  => Komplain::where('status', 'pending')->count(),
        ];

        return view('dashboard', compact('laporans', 'stats'));
    }

    public function histori()
    {
        $user = auth()->user();

        if ($user->isUser()) {
            $laporans = Laporan::where('user_id', $user->id)
                ->whereIn('status', ['selesai', 'ditolak'])
                ->latest()->get();
        } elseif ($user->isStaff()) {
            $laporans = Laporan::whereIn('status', ['selesai', 'ditolak'])
                ->latest()->get();
        } else {
            $laporans = Laporan::whereIn('status', ['selesai', 'ditolak'])
                ->latest()->get();
        }

        return view('histori', compact('laporans'));
    }
}

<?php

namespace App\Http\Controllers;

use App\Models\Laporan;
use App\Models\StatusLaporan;
use Illuminate\Http\Request;

class LaporanController extends Controller
{
    public function create()
    {
        return view('laporan.create');
    }

    public function store(Request $request)
    {
        $data = $request->validate([
            'judul'      => 'required|string|max:150',
            'deskripsi'  => 'required|string',
            'lokasi'     => 'required|string|max:150',
            'foto_bukti' => 'required|image|mimes:jpg,jpeg,png|max:2048',
        ]);

        if ($request->hasFile('foto_bukti')) {
            $data['foto_bukti'] = $request->file('foto_bukti')->store('uploads', 'public');
        }

        $data['user_id']           = auth()->id();
        $data['status']            = 'pending';
        $data['moderation_status'] = 'pending';

        $laporan = Laporan::create($data);

        StatusLaporan::create([
            'laporan_id' => $laporan->id,
            'user_id'    => auth()->id(),
            'status'     => 'pending',
            'keterangan' => 'Laporan baru dibuat, menunggu verifikasi admin.',
        ]);

        return redirect()->route('dashboard')
            ->with('success', 'Laporan berhasil dikirim! Menunggu verifikasi admin.');
    }

    public function show(Laporan $laporan)
    {
        if (auth()->user()->isUser() && $laporan->user_id !== auth()->id()) {
            abort(403);
        }
        return view('laporan.show', compact('laporan'));
    }
}

===== SuperAdminController.php =====
<?php

namespace App\Http\Controllers;

use App\Models\User;
use Illuminate\Http\Request;

class SuperAdminController extends Controller
{
    public function approvals()
    {
        $pendingUsers = User::where('status_akun', 'pending')
            ->where('role', '!=', 'super_admin')
            ->latest()->get();

        $allUsers = User::where('role', '!=', 'super_admin')
            ->latest()->get();

        return view('super.approvals', compact('pendingUsers', 'allUsers'));
    }

    public function approve(User $user)
    {
        if ($user->role === 'super_admin') {
            return back()->with('error', 'Super Admin tidak bisa di-approve.');
        }

        $user->update([
            'status_akun' => 'approved',
            'is_approved' => true,
        ]);

        return back()->with('success', "Akun {$user->name} berhasil disetujui.");
    }

    public function reject(Request $request, User $user)
    {
        if ($user->role === 'super_admin') {
            return back()->with('error', 'Super Admin tidak bisa ditolak.');
        }

        $request->validate([
            'catatan' => 'nullable|string|max:500',
        ]);

        $user->update([
            'status_akun'        => 'rejected',
            'catatan_penolakan'  => $request->catatan,
        ]);

        return back()->with('success', "Pengajuan akun {$user->name} ditolak.");
    }
}

===== RoleMiddleware.php =====
<?php

namespace App\Http\Middleware;

use Closure;
use Illuminate\Http\Request;
use Symfony\Component\HttpFoundation\Response;

class RoleMiddleware
{
    public function handle(Request $request, Closure $next, string ...$roles): Response
    {
        if (!auth()->check()) {
            return redirect()->route('login');
        }

        if (!in_array(auth()->user()->role, $roles)) {
            abort(403, 'Akses ditolak. Anda tidak memiliki izin.');
        }

        return $next($request);
    }
}

===== User.php =====
<?php

namespace App\Models;

use Illuminate\Foundation\Auth\User as Authenticatable;
use Illuminate\Notifications\Notifiable;

class User extends Authenticatable
{
    use Notifiable;

    protected $fillable = [
        'name', 'email', 'password', 'role',
        'nis_nip', 'kelas', 'no_telepon', 'is_approved',
        'alasan_pengajuan', 'status_akun', 'catatan_penolakan',
    ];

    protected $hidden = ['password', 'remember_token'];

    protected $casts = [
        'password'    => 'hashed',
        'is_approved' => 'boolean',
    ];

    public function laporans()
    {
        return $this->hasMany(Laporan::class);
    }

    public function komplains()
    {
        return $this->hasMany(Komplain::class);
    }

    public function isSuperAdmin(): bool { return $this->role === 'super_admin'; }
    public function isAdmin(): bool      { return in_array($this->role, ['admin', 'super_admin']); }
    public function isStaff(): bool      { return $this->role === 'staff'; }
    public function isUser(): bool       { return $this->role === 'user'; }

    public function roleLabel(): string
    {
        return match ($this->role) {
            'super_admin' => 'Super Admin',
            'admin'       => 'Admin',
            'staff'       => 'Staff',
            default       => 'Siswa / Guru',
        };
    }
}

@push('scripts')
<script>
    // Toggle foto perbaikan (wajib kalau status = selesai)
    function toggleFotoPerbaikan() {
        const status = document.getElementById('statusSelect');
        if (!status) return;

        const wrap = document.getElementById('fotoPerbaikanWrap');
        const input = wrap ? wrap.querySelector('input[type=file]') : null;

        if (status.value === 'selesai') {
            wrap.style.display = 'block';
            if (input) input.required = true;
        } else {
            wrap.style.display = 'none';
            if (input) input.required = false;
        }
    }

    document.addEventListener('DOMContentLoaded', function() {
        if (document.getElementById('statusSelect')) {
            toggleFotoPerbaikan();
        }
    });
</script>
@endpush

<script>
    if (typeof mediumZoom === 'function') {
        mediumZoom('.zoomable', { background: 'rgba(0,0,0,.85)' });
    }

    window.LaporAjaStreams = {};
    window.LaporAja_updateBodyState = function () {
        const hasActive = document.querySelector('.laporaja-cam-overlay.active');
        document.body.classList.toggle('cam-active', !!hasActive);
    };

    window.LaporAja_openCamera = function (name) {
        const overlay = document.getElementById('cam_overlay_' + name);
        const video   = document.getElementById('cam_video_' + name);
        const errorEl = document.getElementById('cam_error_' + name);
        const errMsg  = document.getElementById('cam_error_msg_' + name);
        const btnCap  = document.getElementById('cam_capture_' + name);
        if (!overlay) return;

        overlay.classList.add('active');
        document.body.style.overflow = 'hidden';
        window.LaporAja_updateBodyState();

        errorEl.classList.remove('show');
        btnCap.disabled = false;
        video.srcObject = null;

        if (!navigator.mediaDevices || !navigator.mediaDevices.getUserMedia) {
            errorEl.classList.add('show');
            errMsg.textContent = 'Browser tidak mendukung akses kamera.';
            btnCap.disabled = true;
            return;
        }

        navigator.mediaDevices.getUserMedia({
            video: { facingMode: 'environment', width: { ideal: 1280 }, height: { ideal: 720 } },
            audio: false
        })
        .then(function (stream) {
            window.LaporAjaStreams[name] = stream;
            video.srcObject = stream;
            video.play().catch(function () {});
        })
        .catch(function (err) {
            let msg = 'Tidak bisa akses kamera.';
            if (err.name === 'NotAllowedError') msg = 'Izin kamera ditolak. Aktifkan izin kamera.';
            else if (err.name === 'NotFoundError') msg = 'Kamera tidak ditemukan.';
            else if (err.name === 'NotReadableError') msg = 'Kamera sedang dipakai aplikasi lain.';
            errorEl.classList.add('show');
            errMsg.textContent = msg;
            btnCap.disabled = true;
        });
    };

    window.LaporAja_closeCamera = function (name) {
        const overlay = document.getElementById('cam_overlay_' + name);
        const video   = document.getElementById('cam_video_' + name);
        if (window.LaporAjaStreams[name]) {
            window.LaporAjaStreams[name].getTracks().forEach(t => t.stop());
            delete window.LaporAjaStreams[name];
        }
        if (video) video.srcObject = null;
        if (overlay) overlay.classList.remove('active');
        document.body.style.overflow = '';
        window.LaporAja_updateBodyState();
    };

    window.LaporAja_capture = function (name) {
        const video  = document.getElementById('cam_video_' + name);
        const canvas = document.getElementById('cam_canvas_' + name);
        const input  = document.getElementById('input_' + name);
        const prevEl = document.getElementById('preview_' + name);
        const prevWr = document.getElementById('preview_wrap_' + name);
        if (!video.videoWidth) { alert('Kamera belum siap.'); return; }

        canvas.width  = video.videoWidth;
        canvas.height = video.videoHeight;
        canvas.getContext('2d').drawImage(video, 0, 0);

        canvas.toBlob(function (blob) {
            if (!blob) return;
            const file = new File([blob], 'foto_' + Date.now() + '.jpg', { type: 'image/jpeg' });
            const dt = new DataTransfer();
            dt.items.add(file);
            input.files = dt.files;
            prevEl.src = URL.createObjectURL(blob);
            prevWr.style.display = 'block';
            setTimeout(() => { if (typeof mediumZoom === 'function') mediumZoom('.zoomable'); }, 100);
            window.LaporAja_closeCamera(name);
        }, 'image/jpeg', 0.9);
    };

    window.LaporAja_previewFile = function (event, previewId, previewWrapId) {
        const file = event.target.files[0];
        if (!file) return;
        const reader = new FileReader();
        reader.onload = function (e) {
            document.getElementById(previewId).src = e.target.result;
            document.getElementById(previewWrapId).style.display = 'block';
            setTimeout(() => { if (typeof mediumZoom === 'function') mediumZoom('.zoomable'); }, 100);
        };
        reader.readAsDataURL(file);
    };

    document.addEventListener('keydown', function (e) {
        if (e.key === 'Escape') {
            document.querySelectorAll('.laporaja-cam-overlay.active').forEach(function (el) {
                const name = el.id.replace('cam_overlay_', '');
                window.LaporAja_closeCamera(name);
            });
        }
    });
</script>

    // ===== ADMIN & SUPER ADMIN =====
    Route::middleware('role:admin,super_admin')->group(function () {
        // Verifikasi laporan (supervisor)
        Route::get('/admin/verifikasi', [AdminController::class, 'verifikasiIndex'])->name('admin.verifikasi');
        Route::put('/admin/verifikasi/{laporan}', [AdminController::class, 'verifikasi'])->name('admin.verifikasi.proses');

        // Update status
        Route::put('/admin/laporan/{laporan}/status', [AdminController::class, 'updateStatus'])->name('admin.updateStatus');

        // Komplain
        Route::get('/admin/komplain', [AdminController::class, 'komplainIndex'])->name('admin.komplain');
        Route::put('/admin/komplain/{komplain}', [AdminController::class, 'komplainRespon'])->name('admin.komplain.respon');

        // Kelola user

    });

    // ===== SUPER ADMIN =====
    Route::middleware('role:super_admin')->group(function () {
        Route::get('/super/approvals',      [SuperAdminController::class, 'approvals'])->name('super.approvals');
        Route::put('/super/approve/{user}', [SuperAdminController::class, 'approve'])->name('super.approve');

    });
