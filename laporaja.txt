<?php

namespace App\Http\Controllers;

use App\Models\Laporan;
use App\Models\Komplain;
use App\Models\StatusLaporan;
use App\Models\User;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\Hash;
use Illuminate\Support\Facades\Storage;
use Illuminate\Validation\Rule;

class AdminController extends Controller
{
    public function updateStatus(Request $request, Laporan $laporan)
    {
        $rules = [
            'status'  => 'required|in:pending,diproses,selesai,ditolak',
            'catatan' => 'nullable|string|max:1000',
        ];
        if ($request->status === 'selesai') {
            $rules['foto_perbaikan'] = 'required|image|mimes:jpg,jpeg,png|max:2048';
        } else {
            $rules['foto_perbaikan'] = 'nullable|image|mimes:jpg,jpeg,png|max:2048';
        }
        $data = $request->validate($rules, [
            'foto_perbaikan.required' => 'Foto bukti perbaikan wajib diunggah saat status Selesai.',
        ]);
        $update = [
            'status'        => $data['status'],
            'staff_id'      => auth()->id(),
            'catatan_staff' => $data['catatan'] ?? null,
        ];
        if ($request->hasFile('foto_perbaikan')) {
            if ($laporan->foto_perbaikan && Storage::disk('public')->exists($laporan->foto_perbaikan)) {
                Storage::disk('public')->delete($laporan->foto_perbaikan);
            }
            $update['foto_perbaikan'] = $request->file('foto_perbaikan')->store('perbaikan', 'public');
        }
        $laporan->update($update);
        StatusLaporan::create([
            'laporan_id' => $laporan->id,
            'user_id'    => auth()->id(),
            'status'     => $data['status'],
            'keterangan' => $data['catatan'] ?? 'Status diperbarui.',
        ]);
        return back()->with('success', 'Status laporan berhasil diperbarui.');
    }

    public function komplainIndex()
    {
        $komplains = Komplain::latest()->get();
        return view('admin.komplain', compact('komplains'));
    }

    public function komplainRespon(Request $request, Komplain $komplain)
    {
        $data = $request->validate([
            'respon_admin' => 'required|string|max:1000',
            'aksi'         => 'required|in:respon_saja,tolak_laporan,hapus_laporan',
        ]);

        $komplain->update([
            'respon_admin' => $data['respon_admin'],
            'admin_id'     => auth()->id(),
            'respon_at'    => now(),
            'status'       => 'direspon',
        ]);

        if ($data['aksi'] === 'tolak_laporan') {
            $komplain->laporan->update([
                'status'        => 'ditolak',
                'catatan_staff' => 'Ditolak atas permintaan staff: ' . $data['respon_admin'],
            ]);
            StatusLaporan::create([
                'laporan_id' => $komplain->laporan_id,
                'user_id'    => auth()->id(),
                'status'     => 'ditolak',
                'keterangan' => 'Laporan ditolak atas permintaan staff.',
            ]);
            return back()->with('success', 'Laporan ditolak. Tetap tersimpan di histori.');

        } elseif ($data['aksi'] === 'hapus_laporan') {
            $laporan = $komplain->laporan;
            if ($laporan) {
                if ($laporan->foto_bukti && Storage::disk('public')->exists($laporan->foto_bukti)) {
                    Storage::disk('public')->delete($laporan->foto_bukti);
                }
                if ($laporan->foto_perbaikan && Storage::disk('public')->exists($laporan->foto_perbaikan)) {
                    Storage::disk('public')->delete($laporan->foto_perbaikan);
                }
                $laporan->komplains()->delete();
                $laporan->statusLogs()->delete();
                $laporan->delete();
            }
            return back()->with('success', 'Laporan berhasil dihapus permanen.');
        }

        return back()->with('success', 'Respon komplain berhasil dikirim.');
    }

    public function userIndex()
    {
        $users = User::where('role', '!=', 'super_admin')->latest()->get();
        return view('admin.users', compact('users'));
    }

    public function userStore(Request $request)
    {
        $data = $request->validate([
            'name'       => 'required|string|max:100',
            'email'      => 'required|email|unique:users,email',
            'password'   => 'required|min:6',
            'role'       => 'required|in:user,staff,admin',
            'tipe_user'  => 'nullable|in:siswa,guru',
            'nisn_nip'   => 'nullable|string',
            'kelas'      => 'nullable|string',
            'jabatan'    => 'nullable|string',
            'no_telepon' => 'nullable|string',
        ]);

        $data['password'] = Hash::make($data['password']);

        if ($data['role'] === 'user') {
            $data['status_akun'] = 'approved';
            $data['is_approved'] = true;
        } elseif ($data['role'] === 'staff') {
            $data['status_akun'] = 'approved';
            $data['is_approved'] = true;
            $data['tipe_user'] = null;
            $data['kelas']     = null;
        } else {
            $data['status_akun'] = 'pending';
            $data['is_approved'] = false;
            $data['tipe_user'] = null;
            $data['kelas']     = null;
        }

        User::create($data);
        
        $msg = $data['role'] === 'admin' ? "Admin {$data['name']} diajukan. Menunggu approval." : "User {$data['name']} berhasil ditambahkan.";
        return back()->with('success', $msg);
    }

    public function userUpdate(Request $request, User $user)
    {
        // Super Admin tidak bisa diedit oleh Admin biasa
        if ($user->role === 'super_admin' && !auth()->user()->isSuperAdmin()) {
            return back()->with('error', 'Hanya Super Admin yang bisa mengedit Super Admin.');
        }

        $data = $request->validate([
            'name'        => 'required|string|max:100',
            'email'       => ['required', 'email', Rule::unique('users')->ignore($user->id)],
            'role'        => 'required|in:user,staff,admin',
            'tipe_user'   => 'nullable|in:siswa,guru',
            'nisn_nip'    => 'nullable|string',
            'kelas'       => 'nullable|string',
            'jabatan'     => 'nullable|string',
            'no_telepon'  => 'nullable|string',
            'status_akun' => 'required|in:pending,approved,rejected',
        ]);

        if ($data['role'] === 'user') {
            $data['is_approved'] = $data['status_akun'] === 'approved';
        } elseif ($data['role'] === 'staff') {
            $data['status_akun'] = 'approved';
            $data['is_approved'] = true;
            $data['tipe_user'] = null;
            $data['kelas']     = null;
        } else {
            $data['is_approved'] = $data['status_akun'] === 'approved';
            $data['tipe_user'] = null;
            $data['kelas']     = null;
        }
        
        $user->update($data);

        return back()->with('success', "Data user {$user->name} berhasil diperbarui.");
    }

    public function toggleManageUsers(User $user)
    {
        // Hanya Super Admin yang bisa mengakses ini
        if (!auth()->user()->isSuperAdmin()) {
            return back()->with('error', 'Hanya Super Admin yang bisa memberikan izin ini.');
        }

        // Jangan sampai menurunkan derajat Super Admin
        if ($user->role === 'super_admin') {
            return back()->with('error', 'Tidak bisa mengubah izin Super Admin.');
        }

        // Hanya bisa diberikan ke Admin
        if ($user->role !== 'admin') {
            return back()->with('error', 'Izin ini hanya bisa diberikan ke Admin.');
        }

        $user->update([
            'can_manage_users' => !$user->can_manage_users
        ]);

        $status = $user->can_manage_users ? 'diberikan' : 'dicabut';
        return back()->with('success', "Izin kelola user berhasil {$status} untuk {$user->name}.");
    }

    public function userDestroy(User $user)
    {
        if ($user->role === 'super_admin') {
            return back()->with('error', 'Super Admin tidak dapat dihapus.');
        }
        $user->delete();
        return back()->with('success', 'User berhasil dihapus.');
    }
}

===== AuthController.php =====
<?php

namespace App\Http\Controllers;

use App\Models\User;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\Auth;
use Illuminate\Support\Facades\Hash;

class AuthController extends Controller
{
    public function showLogin()
    {
        return view('auth.login');
    }

    public function showRegister()
    {
        return view('auth.register');
    }

    public function register(Request $request)
    {
        $data = $request->validate([
            'name'             => 'required|string|max:100',
            'email'            => 'required|email|unique:users,email',
            'password'         => 'required|min:6|confirmed',
            'nisn_nip'         => 'nullable|string|max:30',
            'kelas'            => 'nullable|string|max:30',
            'no_telepon'       => 'nullable|string|max:20',
            'alasan_pengajuan' => 'nullable|string|max:500',
        ]);

        $data['password']    = Hash::make($data['password']);
        $data['role']        = 'user';
        $data['is_approved'] = false;
        $data['status_akun'] = 'pending';

        User::create($data);

        return redirect()->route('login')
            ->with('success', 'Pengajuan akun berhasil dikirim! Tunggu konfirmasi dari Super Admin.');
    }

    public function login(Request $request)
    {
        $credentials = $request->validate([
            'email'    => 'required|email',
            'password' => 'required',
        ]);

        $user = User::where('email', $credentials['email'])->first();

        if (!$user || !Hash::check($credentials['password'], $user->password)) {
            return back()->withErrors(['email' => 'Email atau password salah.'])->onlyInput('email');
        }

        if ($user->status_akun === 'pending') {
            return back()->withErrors([
                'email' => 'Akun Anda masih menunggu konfirmasi Super Admin.',
            ])->onlyInput('email');
        }

        if ($user->status_akun === 'rejected') {
            $msg = 'Pengajuan akun Anda ditolak.';
            if ($user->catatan_penolakan) {
                $msg .= ' Alasan: ' . $user->catatan_penolakan;
            }
            return back()->withErrors(['email' => $msg])->onlyInput('email');
        }

        if (Auth::attempt($credentials)) {
            $request->session()->regenerate();
            return redirect()->intended(route('dashboard'));
        }

        return back()->withErrors(['email' => 'Email atau password salah.'])->onlyInput('email');
    }

    public function logout(Request $request)
    {
        Auth::logout();
        $request->session()->invalidate();
        $request->session()->regenerateToken();
        return redirect()->route('login');
    }
}

namespace App\Http\Controllers;

use App\Models\Laporan;

class DashboardController extends Controller
{
    public function index()
    {
        $user = auth()->user();

        if ($user->isUser()) {
            $laporans = Laporan::where('user_id', $user->id)->latest()->get();
        } elseif ($user->isStaff()) {
            $laporans = Laporan::whereIn('status', ['pending', 'diproses'])->latest()->get();

===== KomplainController.php =====
<?php

namespace App\Http\Controllers;

use App\Models\Komplain;
use App\Models\Laporan;
use Illuminate\Http\Request;
use Illuminate\Validation\Rule;

class KomplainController extends Controller
{
    public function create(Laporan $laporan)
    {
        return view('komplain.create', compact('laporan'));
    }

    public function store(Request $request, Laporan $laporan)
    {
        $user = auth()->user();

        // Tentukan tipe yang diizinkan berdasarkan role
        if ($user->role === 'user') {
            $allowedTypes = ['pengerjaan_tidak_sesuai', 'lambat_respon', 'lainnya'];
        } elseif ($user->role === 'staff') {
            $allowedTypes = ['kurang_jelas', 'foto_tidak_pantas', 'kata_kasar', 'minta_hapus_laporan', 'lainnya'];
        } else {
            $allowedTypes = ['lainnya'];
        }

        $data = $request->validate([
            'tipe' => ['required', Rule::in($allowedTypes)],
            'isi'  => 'required|string|max:1000',
        ], [
            'tipe.in' => 'Tipe komplain tidak valid untuk role Anda.',
        ]);

        Komplain::create([
            'laporan_id' => $laporan->id,
            'user_id'    => $user->id,
            'tipe'       => $data['tipe'],
            'isi'        => $data['isi'],
            'status'     => 'pending',
        ]);

        return redirect()->route('laporan.show', $laporan->id)
            ->with('success', 'Komplain berhasil dikirim. Admin akan merespon.');
    }

    public function index()
    {
        $komplains = Komplain::where('user_id', auth()->id())
            ->latest()->get();

        // Tandai sudah dibaca
        Komplain::where('user_id', auth()->id())
            ->where('status', 'direspon')
            ->whereNull('read_at')
            ->update(['read_at' => now()]);

        return view('komplain.index', compact('komplains'));
    }
}

namespace App\Http\Controllers;

use App\Models\Laporan;
use App\Models\StatusLaporan;
use Illuminate\Http\Request;

class LaporanController extends Controller
{
    public function create()
    {
        return view('laporan.create');
    }

    public function store(Request $request)
    {
        $data = $request->validate([
            'judul'      => 'required|string|max:150',
            'deskripsi'  => 'required|string',
            'lokasi'     => 'required|string|max:150',
            'foto_bukti' => 'required|image|mimes:jpg,jpeg,png|max:2048',
        ]);

        if ($request->hasFile('foto_bukti')) {
            $data['foto_bukti'] = $request->file('foto_bukti')->store('uploads', 'public');
        }

        $data['user_id'] = auth()->id();
        $data['status']  = 'pending';

        $laporan = Laporan::create($data);

        StatusLaporan::create([
            'laporan_id' => $laporan->id,
            'user_id'    => auth()->id(),
            'status'     => 'pending',
            'keterangan' => 'Laporan baru dibuat.',
        ]);

        return redirect()->route('dashboard')
            ->with('success', 'Laporan berhasil dikirim!');
    }

    public function show(Laporan $laporan)
    {
        if (auth()->user()->isUser() && $laporan->user_id !== auth()->id()) {
            abort(403);
        }
        return view('laporan.show', compact('laporan'));
    }
}

<?php

namespace App\Http\Controllers;

use Illuminate\Http\Request;
use Illuminate\Support\Facades\Hash;
use Illuminate\Validation\Rule;

class ProfileController extends Controller
{
    public function index()
    {
        return view('profile.index');
    }

    public function updateProfile(Request $request)
    {
        $user = auth()->user();

        $data = $request->validate([
            'email'      => ['required', 'email', Rule::unique('users')->ignore($user->id)],
            'kelas'      => 'nullable|string|max:30',
            'jabatan'    => 'nullable|string|max:50',
            'no_telepon' => 'nullable|string|max:20',
        ]);

        $user->update($data);

        return back()->with('success', 'Profil berhasil diperbarui.');
    }

    public function updatePassword(Request $request)
    {
        $request->validate([
            'current_password' => 'required',
            'password'         => 'required|min:6|confirmed',
        ]);

        $user = auth()->user();

        if (!Hash::check($request->current_password, $user->password)) {
            return back()->withErrors(['current_password' => 'Password saat ini tidak cocok.']);
        }

        $user->update([
            'password' => Hash::make($request->password),
        ]);

        return back()->with('success', 'Password berhasil diubah.');
    }
}

===== SuperAdminController.php =====
<?php

namespace App\Http\Controllers;

use App\Models\User;
use Illuminate\Http\Request;

class SuperAdminController extends Controller
{
    public function approvals()
    {
        $pendingUsers = User::where('status_akun', 'pending')
            ->where('role', '!=', 'super_admin')
            ->latest()->get();

        $allUsers = User::where('role', '!=', 'super_admin')
            ->latest()->get();

        return view('super.approvals', compact('pendingUsers', 'allUsers'));
    }

    public function approve(User $user)
    {
        if ($user->role === 'super_admin') {
            return back()->with('error', 'Super Admin tidak bisa di-approve.');
        }

        $user->update([
            'status_akun' => 'approved',
            'is_approved' => true,
        ]);

        return back()->with('success', "Akun {$user->name} berhasil disetujui.");
    }

    public function reject(Request $request, User $user)
    {
        if ($user->role === 'super_admin') {
            return back()->with('error', 'Super Admin tidak bisa ditolak.');
        }

        $request->validate([
            'catatan' => 'nullable|string|max:500',
        ]);

        $user->update([
            'status_akun'        => 'rejected',
            'catatan_penolakan'  => $request->catatan,
        ]);

        return back()->with('success', "Pengajuan akun {$user->name} ditolak.");
    }
}

===== CanManageUsers.php =====
<?php

namespace App\Http\Middleware;

use Closure;
use Illuminate\Http\Request;
use Symfony\Component\HttpFoundation\Response;

class CanManageUsers
{
    public function handle(Request $request, Closure $next): Response
    {
        $user = auth()->user();

        // Super Admin selalu bisa, atau Admin yang punya izin can_manage_users
        if ($user->isSuperAdmin() || ($user->isAdmin() && $user->can_manage_users)) {
            return $next($request);
        }

        abort(403, 'Anda tidak memiliki izin untuk mengelola user.');
    }
}

===== RoleMiddleware.php =====
<?php

namespace App\Http\Middleware;

use Closure;
use Illuminate\Http\Request;
use Symfony\Component\HttpFoundation\Response;

class RoleMiddleware
{
    public function handle(Request $request, Closure $next, string ...$roles): Response
    {
        if (!auth()->check()) {
            return redirect()->route('login');
        }

        if (!in_array(auth()->user()->role, $roles)) {
            abort(403, 'Akses ditolak. Anda tidak memiliki izin.');
        }

        return $next($request);
    }
}

===== User.php =====
<?php

namespace App\Models;

use Illuminate\Foundation\Auth\User as Authenticatable;
use Illuminate\Notifications\Notifiable;

class User extends Authenticatable
{
    use Notifiable;

    protected $fillable = [
        'name', 'email', 'password', 'role', 'tipe_user', 'can_manage_users',
        'nisn_nip', 'kelas', 'jabatan', 'no_telepon', 'is_approved',
        'alasan_pengajuan', 'status_akun', 'catatan_penolakan',
    ];

    protected $hidden = ['password', 'remember_token'];

    protected $casts = [
        'password'         => 'hashed',
        'is_approved'      => 'boolean',
        'can_manage_users' => 'boolean',
    ];

    public function laporans()
    {
        return $this->hasMany(Laporan::class);
    }

    public function komplains()
    {
        return $this->hasMany(Komplain::class);
    }

    public function isSuperAdmin(): bool { return $this->role === 'super_admin'; }
    public function isAdmin(): bool      { return in_array($this->role, ['admin', 'super_admin']); }
    public function isStaff(): bool      { return $this->role === 'staff'; }
    public function isUser(): bool       { return $this->role === 'user'; }

    public function roleLabel(): string
    {
        if ($this->role === 'user') {
            return $this->tipe_user === 'guru' ? 'Guru' : 'Siswa';
        }
        return match ($this->role) {
            'super_admin' => 'Super Admin',
            'admin'       => 'Admin',

<?php

namespace Database\Factories;

use App\Models\User;
use Illuminate\Database\Eloquent\Factories\Factory;
use Illuminate\Support\Facades\Hash;
use Illuminate\Support\Str;

/**
 * @extends Factory<User>
 */
class UserFactory extends Factory
{
    /**
     * The current password being used by the factory.
     */
    protected static ?string $password;

    /**
     * Define the model's default state.
     *
     * @return array<string, mixed>
     */
    public function definition(): array
    {
        return [
            'name' => fake()->name(),
            'email' => fake()->unique()->safeEmail(),
            'email_verified_at' => now(),
            'password' => static::$password ??= Hash::make('password'),
            'remember_token' => Str::random(10),
        ];
    }

    /**
     * Indicate that the model's email address should be unverified.
     */
    public function unverified(): static
    {
        return $this->state(fn (array $attributes) => [
            'email_verified_at' => null,
        ]);
    }
}

<script>
    function updateForm(prefix) {
        const roleEl = document.getElementById('role-' + prefix);
        if (!roleEl) return;
        
        const role = roleEl.value;
        const wrapTipe = document.getElementById('wrap-tipe-' + prefix);
        const wrapStatus = document.getElementById('wrap-status-' + prefix);
        
        document.querySelectorAll(`#form-siswa-${prefix}, #form-guru-${prefix}, #form-pegawai-${prefix}`).forEach(el => el.style.display = 'none');
        document.querySelectorAll(`#form-siswa-${prefix} input, #form-guru-${prefix} input, #form-pegawai-${prefix} input`).forEach(i => i.disabled = true);

        if (role === 'user') {
            if(wrapTipe) wrapTipe.style.display = 'block';
            if(wrapStatus) wrapStatus.style.display = 'block';
            
            const tipe = document.getElementById('tipe-' + prefix).value;
            if (tipe === 'siswa') {
                document.getElementById(`form-siswa-${prefix}`).style.display = 'block';
                document.querySelectorAll(`#form-siswa-${prefix} input`).forEach(i => i.disabled = false);
            } else {
                document.getElementById(`form-guru-${prefix}`).style.display = 'block';
                document.querySelectorAll(`#form-guru-${prefix} input`).forEach(i => i.disabled = false);
            }
        } else if (role === 'staff') {
            if(wrapTipe) wrapTipe.style.display = 'none';
            if(wrapStatus) wrapStatus.style.display = 'none'; 
            document.getElementById(`form-pegawai-${prefix}`).style.display = 'block';
            document.querySelectorAll(`#form-pegawai-${prefix} input`).forEach(i => i.disabled = false);
        } else if (role === 'admin') {
            if(wrapTipe) wrapTipe.style.display = 'none';
            if(wrapStatus) wrapStatus.style.display = 'block';
            document.getElementById(`form-pegawai-${prefix}`).style.display = 'block';
            document.querySelectorAll(`#form-pegawai-${prefix} input`).forEach(i => i.disabled = false);
        } else {
            if(wrapTipe) wrapTipe.style.display = 'none';
            if(wrapStatus) wrapStatus.style.display = 'block';
        }

        if (prefix === 'add') {
            const warning = document.getElementById('adminWarning');
            if (warning) {
                if (role === 'admin') warning.classList.remove('d-none');
                else warning.classList.add('d-none');
            }
        }
    }

    document.addEventListener('DOMContentLoaded', () => {
        updateForm('add');
        @foreach($users as $u)
            updateForm('{{ $u->id }}');
        @endforeach
    });
</script>
@endpush

<script>
    function toggleFotoPerbaikan() {
        const status = document.getElementById('statusSelect');
        const wrap = document.getElementById('fotoPerbaikanWrap');
        const input = wrap ? wrap.querySelector('input[type=file]') : null;
        if (status.value === 'selesai') {
            wrap.style.display = 'block';
            if (input) input.required = true;
        } else {
            wrap.style.display = 'none';
            if (input) input.required = false;
        }
    }
</script>

<script>
    if (typeof mediumZoom === 'function') {
        mediumZoom('.zoomable', { background: 'rgba(0,0,0,.85)' });
    }

    window.LaporAjaStreams = {};
    window.LaporAja_updateBodyState = function () {
        const hasActive = document.querySelector('.laporaja-cam-overlay.active');
        document.body.classList.toggle('cam-active', !!hasActive);
    };

    window.LaporAja_openCamera = function (name) {
        const overlay = document.getElementById('cam_overlay_' + name);
        const video   = document.getElementById('cam_video_' + name);
        const errorEl = document.getElementById('cam_error_' + name);
        const errMsg  = document.getElementById('cam_error_msg_' + name);
        const btnCap  = document.getElementById('cam_capture_' + name);
        if (!overlay) return;

        overlay.classList.add('active');
        document.body.style.overflow = 'hidden';
        window.LaporAja_updateBodyState();

        errorEl.classList.remove('show');
        btnCap.disabled = false;
        video.srcObject = null;

        if (!navigator.mediaDevices || !navigator.mediaDevices.getUserMedia) {
            errorEl.classList.add('show');
            errMsg.textContent = 'Browser tidak mendukung akses kamera.';
            btnCap.disabled = true;
            return;
        }

        navigator.mediaDevices.getUserMedia({
            video: { facingMode: 'environment', width: { ideal: 1280 }, height: { ideal: 720 } },
            audio: false
        })
        .then(function (stream) {
            window.LaporAjaStreams[name] = stream;
            video.srcObject = stream;
            video.play().catch(function () {});
        })
        .catch(function (err) {
            let msg = 'Tidak bisa akses kamera.';
            if (err.name === 'NotAllowedError') msg = 'Izin kamera ditolak. Aktifkan izin kamera.';
            else if (err.name === 'NotFoundError') msg = 'Kamera tidak ditemukan.';
            else if (err.name === 'NotReadableError') msg = 'Kamera sedang dipakai aplikasi lain.';
            errorEl.classList.add('show');
            errMsg.textContent = msg;
            btnCap.disabled = true;
        });
    };

    window.LaporAja_closeCamera = function (name) {
        const overlay = document.getElementById('cam_overlay_' + name);
        const video   = document.getElementById('cam_video_' + name);
        if (window.LaporAjaStreams[name]) {
            window.LaporAjaStreams[name].getTracks().forEach(t => t.stop());
            delete window.LaporAjaStreams[name];
        }
        if (video) video.srcObject = null;
        if (overlay) overlay.classList.remove('active');
        document.body.style.overflow = '';
        window.LaporAja_updateBodyState();
    };

    window.LaporAja_capture = function (name) {
        const video  = document.getElementById('cam_video_' + name);
        const canvas = document.getElementById('cam_canvas_' + name);
        const input  = document.getElementById('input_' + name);
        const prevEl = document.getElementById('preview_' + name);
        const prevWr = document.getElementById('preview_wrap_' + name);
        if (!video.videoWidth) { alert('Kamera belum siap.'); return; }

        canvas.width  = video.videoWidth;
        canvas.height = video.videoHeight;
        canvas.getContext('2d').drawImage(video, 0, 0);

        canvas.toBlob(function (blob) {
            if (!blob) return;
            const file = new File([blob], 'foto_' + Date.now() + '.jpg', { type: 'image/jpeg' });
            const dt = new DataTransfer();
            dt.items.add(file);
            input.files = dt.files;
            prevEl.src = URL.createObjectURL(blob);
            prevWr.style.display = 'block';
            setTimeout(() => { if (typeof mediumZoom === 'function') mediumZoom('.zoomable'); }, 100);
            window.LaporAja_closeCamera(name);
        }, 'image/jpeg', 0.9);
    };

    window.LaporAja_previewFile = function (event, previewId, previewWrapId) {
        const file = event.target.files[0];
        if (!file) return;
        const reader = new FileReader();
        reader.onload = function (e) {
            document.getElementById(previewId).src = e.target.result;
            document.getElementById(previewWrapId).style.display = 'block';
            setTimeout(() => { if (typeof mediumZoom === 'function') mediumZoom('.zoomable'); }, 100);
        };
        reader.readAsDataURL(file);
    };

    document.addEventListener('keydown', function (e) {
        if (e.key === 'Escape') {
            document.querySelectorAll('.laporaja-cam-overlay.active').forEach(function (el) {
                const name = el.id.replace('cam_overlay_', '');
                window.LaporAja_closeCamera(name);
            });
        }
    });

===== routes/web.php =====
<?php

use App\Http\Controllers\{
    AuthController,
    DashboardController,
    LaporanController,
    AdminController,
    StaffController,
    SuperAdminController,
    KomplainController,
    ProfileController,
};
use Illuminate\Support\Facades\Route;

Route::get('/', fn() => redirect()->route('login'));

// ============ AUTH ============
Route::get('/login',    [AuthController::class, 'showLogin'])->name('login');
Route::post('/login',   [AuthController::class, 'login']);
Route::get('/register', [AuthController::class, 'showRegister'])->name('register');
Route::post('/register',[AuthController::class, 'register']);
Route::post('/logout',  [AuthController::class, 'logout'])->name('logout');

// ============ LOGIN REQUIRED ============
Route::middleware('auth')->group(function () {

    Route::get('/dashboard', [DashboardController::class, 'index'])->name('dashboard');
    Route::get('/histori',   [DashboardController::class, 'histori'])->name('histori');

    // ===== PROFIL =====
    Route::get('/profile', [ProfileController::class, 'index'])->name('profile.index');
    Route::put('/profile/update', [ProfileController::class, 'updateProfile'])->name('profile.update');
    Route::put('/profile/password', [ProfileController::class, 'updatePassword'])->name('profile.password');

    // ===== USER =====
    Route::middleware('role:user')->group(function () {
        Route::get('/laporan/create', [LaporanController::class, 'create'])->name('laporan.create');
        Route::post('/laporan',       [LaporanController::class, 'store'])->name('laporan.store');
    });

    // ===== KOMPLAIN =====
    Route::middleware('role:user,staff')->group(function () {
        Route::get('/laporan/{laporan}/komplain',  [KomplainController::class, 'create'])->name('komplain.create');
        Route::post('/laporan/{laporan}/komplain', [KomplainController::class, 'store'])->name('komplain.store');
        Route::get('/komplain-saya',               [KomplainController::class, 'index'])->name('komplain.index');
    });

    // Detail laporan (semua role)
    Route::get('/laporan/{laporan}', [LaporanController::class, 'show'])->name('laporan.show');

    // ===== STAFF =====
    Route::middleware('role:staff')->group(function () {
        Route::get('/staff/laporan', [StaffController::class, 'index'])->name('staff.laporan');
        Route::put('/staff/laporan/{laporan}/status', [StaffController::class, 'updateStatus'])->name('staff.updateStatus');
    });

    // ===== ADMIN & SUPER ADMIN =====
    Route::middleware('role:admin,super_admin')->group(function () {
        Route::put('/admin/laporan/{laporan}/status', [AdminController::class, 'updateStatus'])->name('admin.updateStatus');
        Route::get('/admin/komplain', [AdminController::class, 'komplainIndex'])->name('admin.komplain');
        Route::put('/admin/komplain/{komplain}', [AdminController::class, 'komplainRespon'])->name('admin.komplain.respon');

        // Kelola user (Dibatasi middleware can_manage_users)
        Route::middleware('can_manage_users')->group(function () {
            Route::get('/admin/users', [AdminController::class, 'userIndex'])->name('admin.users');
            Route::post('/admin/users', [AdminController::class, 'userStore'])->name('admin.users.store');
            Route::put('/admin/users/{user}', [AdminController::class, 'userUpdate'])->name('admin.users.update');
            Route::delete('/admin/users/{user}', [AdminController::class, 'userDestroy'])->name('admin.users.destroy');
            Route::put('/admin/users/{user}/toggle-manage', [AdminController::class, 'toggleManageUsers'])->name('admin.users.toggle_manage');
        });
    });

    // ===== SUPER ADMIN =====
    Route::middleware('role:super_admin')->group(function () {
        Route::get('/super/approvals',      [SuperAdminController::class, 'approvals'])->name('super.approvals');
        Route::put('/super/approve/{user}', [SuperAdminController::class, 'approve'])->name('super.approve');
        Route::put('/super/reject/{user}',  [SuperAdminController::class, 'reject'])->name('super.reject');
    });
});
